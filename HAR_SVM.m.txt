clc;
clear;
close all;

%% Load UCI HAR Feature Data

X_train = load('UCI HAR Dataset/train/X_train.txt');
Y_train = load('UCI HAR Dataset/train/y_train.txt');

X_test = load('UCI HAR Dataset/test/X_test.txt');
Y_test = load('UCI HAR Dataset/test/y_test.txt');

%% Normalize Features Using Training Data

feature_mean = mean(X_train);
feature_std = std(X_train);

X_train = (X_train - feature_mean) ./ feature_std;
X_test = (X_test - feature_mean) ./ feature_std;

%% Create RBF SVM

SVM_Template = templateSVM( ...
    'KernelFunction','rbf', ...
    'KernelScale','auto', ...
    'BoxConstraint',10, ...
    'Standardize',false);

%% Train Multiclass SVM

SVM_Model = fitcecoc( ...
    X_train, ...
    Y_train, ...
    'Learners',SVM_Template, ...
    'Coding','onevsone');

%% Predict Test Data

Y_pred = predict(SVM_Model, X_test);

%% Calculate Accuracy

accuracy = sum(Y_pred == Y_test) / numel(Y_test) * 100;
fprintf('\nSVM Classification Accuracy = %.2f%%\n', accuracy); %[output:94a4735f]

%% Confusion Matrix

classOrder = 1:6;
cm = confusionmat(Y_test, Y_pred, 'Order', classOrder);

figure; %[output:905d3663]
confusionchart(cm, classOrder); %[output:905d3663]
title('RBF SVM Classification Confusion Matrix'); %[output:905d3663]

%% Calculate F1 Score

truePositive = diag(cm);
precision = truePositive ./ sum(cm, 1)';
recall = truePositive ./ sum(cm, 2);

f1PerClass = zeros(size(truePositive));
validIdx = (precision + recall) > 0;
f1PerClass(validIdx) = 2 * precision(validIdx) .* recall(validIdx) ./ ...
    (precision(validIdx) + recall(validIdx));

macroF1 = mean(f1PerClass);

fprintf('Macro F1 Score = %.4f\n', macroF1); %[output:41cbf951]

%% Activity Names

classNames = {
    'WALKING'
    'WALKING UPSTAIRS'
    'WALKING DOWNSTAIRS'
    'SITTING'
    'STANDING'
    'LAYING'
};

disp(' '); %[output:2fd37a0d]
disp('Activity Class Mapping:'); %[output:49d59228]

for i = 1:6 %[output:group:228b01d6]
    fprintf('%d = %s\n', i, classNames{i}); %[output:4b54e33f]
end %[output:group:228b01d6]
